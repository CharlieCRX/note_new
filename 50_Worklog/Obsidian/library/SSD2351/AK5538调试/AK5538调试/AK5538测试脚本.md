```
#!/bin/sh

# SPDX-License-Identifier: GPL-2.0

#

# Initialize the USBL/PCUPID AK5538 from userspace for the validated

# 192 kHz, stereo, 32-bit normal-I2S capture mode.

#

# Physical clock ownership:

#   FPGA   -> AK5538 MCLK (24.576 MHz)

#   AK5538 -> CPU BCLK/LRCK (codec provider, CPU slave)

#

# Dependencies:

#   fpga, i2cget, i2cset, /dev/i2c-1

#

# After this script succeeds, capture with:

#   arecord -D hw:0,1 -t wav -f S32_LE -r 192000 -c 2 -d 5 output.wav

set -eu

I2C_BUS=1
I2C_DEV="/dev/i2c-${I2C_BUS}"
AK5538_ADDR=0x10

FPGA_REG_AD_CLK_SEL=0x2
FPGA_REG_RST_CTL=0x5
FPGA_RST_AD_ASSERT=0x1B
FPGA_RST_AD_DEASSERT=0x1F
FPGA_MCLK_24P576M=7

AK5538_REG_POWER_MGMT1=0x00
AK5538_REG_POWER_MGMT2=0x01
AK5538_REG_CONTROL1=0x02
AK5538_REG_CONTROL2=0x03
AK5538_REG_CONTROL3=0x04

AK5538_POWER_ALL_CHANNELS=0xFF
AK5538_RST_ASSERT=0x00
AK5538_RST_DEASSERT=0x01
AK5538_CONTROL1_192K_S32_I2S_HPF=0x07
AK5538_CONTROL2_NORMAL_I2S=0x00
AK5538_CONTROL3_PCM_SHARP=0x00

CAPTURE_NODE=/dev/snd/pcmC0D1c
LOCK_DIR=/tmp/ak5538-userspace-init.lock

die()
{
    echo "[FAIL] $*" >&2
    exit 1
}

cleanup()
{
    rmdir "${LOCK_DIR}" 2>/dev/null || true
}

need_command()
{
    command -v "$1" >/dev/null 2>&1 || die "missing command: $1"
}

write_ak5538()
{
    reg=$1
    value=$2

    echo "  i2cset reg=${reg} value=${value}"
    i2cset -y "${I2C_BUS}" "${AK5538_ADDR}" "${reg}" "${value}"

}

read_ak5538()
{
    reg=$1
    i2cget -y "${I2C_BUS}" "${AK5538_ADDR}" "${reg}"
}

verify_ak5538()
{
    reg=$1
    expected=$2
    name=$3
    actual=$(read_ak5538 "${reg}") || die "cannot read ${name} (${reg})"

    if [ "$((actual))" -ne "$((expected))" ]; then
        die "${name} mismatch: expected ${expected}, got ${actual}"
    fi
    
    echo "  ${name} (${reg}) = ${actual} [OK]"

}

if ! mkdir "${LOCK_DIR}" 2>/dev/null; then
    die "another AK5538 initialization is running (${LOCK_DIR})"
fi
trap cleanup 0
trap 'exit 130' 1 2 15

need_command fpga
need_command i2cget
need_command i2cset

[ -e "${I2C_DEV}" ] || die "${I2C_DEV} does not exist"

if [ -e "/sys/bus/i2c/devices/${I2C_BUS}-0010" ]; then
    die "I2C address 0x10 is still registered by the kernel; check the DTB"
fi

if command -v fuser >/dev/null 2>&1 && [ -e "${CAPTURE_NODE}" ]; then
    if fuser "${CAPTURE_NODE}" >/dev/null 2>&1; then
        die "${CAPTURE_NODE} is busy; stop arecord before changing clocks"
    fi
fi

echo "=================================================="
echo " AK5538 userspace initialization"
echo " mode : 192000 Hz, 2 channels, S32_LE, normal I2S"
echo " MCLK : 24.576 MHz (FPGA register value 7)"
echo "=================================================="

echo "[1/7] Assert external AK5538 reset"
fpga -w "${FPGA_REG_RST_CTL}" "${FPGA_RST_AD_ASSERT}"
sleep 0.1

echo "[2/7] Select 24.576 MHz FPGA MCLK while reset is asserted"
fpga -w "${FPGA_REG_AD_CLK_SEL}" "${FPGA_MCLK_24P576M}"
echo -n "  FPGA clock register = "
fpga -r "${FPGA_REG_AD_CLK_SEL}"
sleep 0.1

echo "[3/7] Deassert external AK5538 reset"
fpga -w "${FPGA_REG_RST_CTL}" "${FPGA_RST_AD_DEASSERT}"
echo -n "  FPGA reset register = "
fpga -r "${FPGA_REG_RST_CTL}"
sleep 0.1

echo "[4/7] Assert AK5538 internal timing reset"
write_ak5538 "${AK5538_REG_POWER_MGMT2}" "${AK5538_RST_ASSERT}"

echo "[5/7] Configure AK5538 PCM and serial interface"
write_ak5538 "${AK5538_REG_POWER_MGMT1}" "${AK5538_POWER_ALL_CHANNELS}"
write_ak5538 "${AK5538_REG_CONTROL1}" "${AK5538_CONTROL1_192K_S32_I2S_HPF}"
write_ak5538 "${AK5538_REG_CONTROL2}" "${AK5538_CONTROL2_NORMAL_I2S}"
write_ak5538 "${AK5538_REG_CONTROL3}" "${AK5538_CONTROL3_PCM_SHARP}"

echo "[6/7] Release AK5538 internal timing reset"
write_ak5538 "${AK5538_REG_POWER_MGMT2}" "${AK5538_RST_DEASSERT}"

# The AK5538 requires at least 578/fs for valid data. 20 ms is safely

# above that requirement at 192 kHz and is still short for a test tool.

sleep 0.02

echo "[7/7] Verify AK5538 register values"
verify_ak5538 "${AK5538_REG_POWER_MGMT1}" "${AK5538_POWER_ALL_CHANNELS}" "POWER_MGMT1"
verify_ak5538 "${AK5538_REG_POWER_MGMT2}" "${AK5538_RST_DEASSERT}" "POWER_MGMT2"
verify_ak5538 "${AK5538_REG_CONTROL1}" "${AK5538_CONTROL1_192K_S32_I2S_HPF}" "CONTROL1"
verify_ak5538 "${AK5538_REG_CONTROL2}" "${AK5538_CONTROL2_NORMAL_I2S}" "CONTROL2"
verify_ak5538 "${AK5538_REG_CONTROL3}" "${AK5538_CONTROL3_PCM_SHARP}" "CONTROL3"

echo "=================================================="
echo " AK5538 initialization completed successfully"
echo "=================================================="
echo "Capture command:"
echo "  arecord -D hw:0,1 -t wav -f S32_LE -r 192000 -c 2 -d 5 /mnt/ak5538-192k-stereo.wav"
```
